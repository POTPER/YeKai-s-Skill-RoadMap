# WebGPU 纹理和图像处理完整指南

## 纹理基础知识

### 什么是纹理？
纹理是存储在 GPU 内存中的多维数组，主要用于：
- **图像存储**: 存储 2D 图像数据
- **表面贴图**: 为 3D 模型添加表面细节
- **数据存储**: 存储任意的数值数据
- **渲染目标**: 作为渲染的输出目标

## 纹理操作类型

### 1. 纹理采样（Sampling）

```wgsl

@group(0) @binding(1) var mySampler: sampler;

@fragment
fn fs_main(@location(0) uv: vec2<f32>) -> @location(0) vec4<f32> {
    // 采样纹理

    return color;
}
```

## 图像滤镜实现

### 1. 高斯模糊
```wgsl
@compute @workgroup_size(8, 8)
fn gaussianBlur(@builtin(global_invocation_id) global_id: vec3<u32>) {
    let coord = vec2<i32>(global_id.xy);
    let dimensions = textureDimensions(inputTexture);
    
    if (coord.x >= dimensions.x || coord.y >= dimensions.y) {
        return;
    }
    
    // 5x5 高斯核
    let kernel = array<array<f32, 5>, 5>(
        array<f32, 5>(1.0, 4.0, 6.0, 4.0, 1.0),
        array<f32, 5>(4.0, 16.0, 24.0, 16.0, 4.0),
        array<f32, 5>(6.0, 24.0, 36.0, 24.0, 6.0),
        array<f32, 5>(4.0, 16.0, 24.0, 16.0, 4.0),
        array<f32, 5>(1.0, 4.0, 6.0, 4.0, 1.0)
    );
    
    var sum = vec4<f32>(0.0);
    var weightSum = 0.0;
    
    for (var y = -2; y <= 2; y++) {
        for (var x = -2; x <= 2; x++) {
            let sampleCoord = coord + vec2<i32>(x, y);
            if (sampleCoord.x >= 0 && sampleCoord.x < dimensions.x &&
                sampleCoord.y >= 0 && sampleCoord.y < dimensions.y) {
                
                let weight = kernel[y + 2][x + 2] / 256.0;
                let color = textureLoad(inputTexture, sampleCoord, 0);
                sum += color * weight;
                weightSum += weight;
            }
        }
    }
    
    textureStore(outputTexture, coord, sum / weightSum);
}
```

### 2. 边缘检测（Sobel 算子）
```wgsl
@compute @workgroup_size(8, 8)
fn sobelEdgeDetection(@builtin(global_invocation_id) global_id: vec3<u32>) {
    let coord = vec2<i32>(global_id.xy);
    let dimensions = textureDimensions(inputTexture);
    
    if (coord.x >= dimensions.x || coord.y >= dimensions.y) {
        return;
    }
    
    // Sobel X 核
    let sobelX = array<array<f32, 3>, 3>(
        array<f32, 3>(-1.0, 0.0, 1.0),
        array<f32, 3>(-2.0, 0.0, 2.0),
        array<f32, 3>(-1.0, 0.0, 1.0)
    );
    
    // Sobel Y 核
    let sobelY = array<array<f32, 3>, 3>(
        array<f32, 3>(-1.0, -2.0, -1.0),
        array<f32, 3>( 0.0,  0.0,  0.0),
        array<f32, 3>( 1.0,  2.0,  1.0)
    );
    
    var gx = 0.0;
    var gy = 0.0;
    
    for (var y = -1; y <= 1; y++) {
        for (var x = -1; x <= 1; x++) {
            let sampleCoord = coord + vec2<i32>(x, y);
            let color = textureLoad(inputTexture, sampleCoord, 0);
            let gray = dot(color.rgb, vec3<f32>(0.299, 0.587, 0.114));
            
            gx += gray * sobelX[y + 1][x + 1];
            gy += gray * sobelY[y + 1][x + 1];
        }
    }
    
    let magnitude = sqrt(gx * gx + gy * gy);
    textureStore(outputTexture, coord, vec4<f32>(magnitude, magnitude, magnitude, 1.0));
}
```

### 3. 颜色调整
```wgsl
struct ColorAdjustment {
    brightness: f32,
    contrast: f32,
    saturation: f32,
    hue: f32,
}

@group(0) @binding(2) var<uniform> adjustment: ColorAdjustment;

@compute @workgroup_size(8, 8)
fn colorAdjust(@builtin(global_invocation_id) global_id: vec3<u32>) {
    let coord = vec2<i32>(global_id.xy);
    var color = textureLoad(inputTexture, coord, 0);
    
    // 亮度调整
    color.rgb = color.rgb + adjustment.brightness;
    
    // 对比度调整
    color.rgb = (color.rgb - 0.5) * adjustment.contrast + 0.5;
    
    // 饱和度调整
    let gray = dot(color.rgb, vec3<f32>(0.299, 0.587, 0.114));
    color.rgb = mix(vec3<f32>(gray), color.rgb, adjustment.saturation);
    
    // 色相调整
    color.rgb = hueShift(color.rgb, adjustment.hue);
    
    textureStore(outputTexture, coord, clamp(color, vec4<f32>(0.0), vec4<f32>(1.0)));
}

// 色相偏移函数
fn hueShift(color: vec3<f32>, hue: f32) -> vec3<f32> {
    let cosA = cos(hue);
    let sinA = sin(hue);
    
    let matrix = mat3x3<f32>(
        cosA + (1.0 - cosA) / 3.0,
        1.0/3.0 * (1.0 - cosA) - sqrt(1.0/3.0) * sinA,
        1.0/3.0 * (1.0 - cosA) + sqrt(1.0/3.0) * sinA,
        
        1.0/3.0 * (1.0 - cosA) + sqrt(1.0/3.0) * sinA,
        cosA + 1.0/3.0 * (1.0 - cosA),
        1.0/3.0 * (1.0 - cosA) - sqrt(1.0/3.0) * sinA,
        
        1.0/3.0 * (1.0 - cosA) - sqrt(1.0/3.0) * sinA,
        1.0/3.0 * (1.0 - cosA) + sqrt(1.0/3.0) * sinA,
        cosA + 1.0/3.0 * (1.0 - cosA)
    );
    
    return matrix * color;
}
```

```javascript
function createVideoTexture(video) {
    return device.createTexture({
        size: [video.videoWidth, video.videoHeight],
        format: 'rgba8unorm',
        usage: GPUTextureUsage.TEXTURE_BINDING | 
               GPUTextureUsage.COPY_DST |
               GPUTextureUsage.RENDER_ATTACHMENT
    });
}

// 实时处理循环
function processVideoFrame(video, inputTexture, outputTexture) {
    // 将视频帧复制到纹理
    device.queue.copyExternalImageToTexture(
        { source: video },
        { texture: inputTexture },
        [video.videoWidth, video.videoHeight]
    );
    
    // 应用图像处理
    const commandEncoder = device.createCommandEncoder();
    const computePass = commandEncoder.beginComputePass();
    computePass.setPipeline(filterPipeline);
    computePass.setBindGroup(0, bindGroup);
    computePass.dispatchWorkgroups(
        Math.ceil(video.videoWidth / 8),
        Math.ceil(video.videoHeight / 8)
    );
    computePass.end();
    
    device.queue.submit([commandEncoder.finish()]);
    
    requestAnimationFrame(() => processVideoFrame(video, inputTexture, outputTexture));
}
```

## 3D 纹理映射

### UV 坐标和纹理采样
```wgsl
struct VertexInput {
    @location(0) position: vec3<f32>,
    @location(1) normal: vec3<f32>,
    @location(2) uv: vec2<f32>,
}

struct VertexOutput {
    @builtin(position) position: vec4<f32>,
    @location(0) worldPos: vec3<f32>,
    @location(1) normal: vec3<f32>,
    @location(2) uv: vec2<f32>,
}

@group(1) @binding(0) var diffuseTexture: texture_2d<f32>;
@group(1) @binding(1) var normalTexture: texture_2d<f32>;
@group(1) @binding(2) var textureSampler: sampler;

@fragment
fn fs_main(input: VertexOutput) -> @location(0> vec4<f32> {
    // 漫反射纹理采样
    let baseColor = textureSample(diffuseTexture, textureSampler, input.uv);
    
    // 法线贴图采样
    let normalMap = textureSample(normalTexture, textureSampler, input.uv);
    let normal = normalize(normalMap.xyz * 2.0 - 1.0);
    
    // 简单光照计算
    let lightDir = normalize(vec3<f32>(1.0, 1.0, 1.0));
    let ndotl = max(dot(normal, lightDir), 0.0);
    
    return vec4<f32>(baseColor.rgb * ndotl, baseColor.a);
}
```

### 2. 阴影映射
```wgsl
@group(2) @binding(0) var shadowMap: texture_depth_2d;
@group(2) @binding(1) var shadowSampler: sampler_comparison;

fn calculateShadow(lightSpacePos: vec4<f32>) -> f32 {
    let projCoords = lightSpacePos.xyz / lightSpacePos.w;
    let shadowCoords = projCoords * 0.5 + 0.5;
    
    if (shadowCoords.x < 0.0 || shadowCoords.x > 1.0 ||
        shadowCoords.y < 0.0 || shadowCoords.y > 1.0 ||
        shadowCoords.z < 0.0 || shadowCoords.z > 1.0) {
        return 1.0;
    }
    
    return textureSampleCompare(shadowMap, shadowSampler, 
                               shadowCoords.xy, shadowCoords.z);
}
```

### 3. 纹理流式加载
```javascript
class TextureStreamer {
    constructor(device) {
        this.device = device;
        this.loadQueue = [];
        this.cache = new Map();
    }
    
    async loadTexture(url, priority = 0) {
        if (this.cache.has(url)) {
            return this.cache.get(url);
        }
        
        return new Promise((resolve) => {
            this.loadQueue.push({ url, priority, resolve });
            this.loadQueue.sort((a, b) => b.priority - a.priority);
            this.processQueue();
        });
    }
    
    async processQueue() {
        if (this.loadQueue.length === 0) return;
        
        const { url, resolve } = this.loadQueue.shift();
        const texture = await this.createTextureFromURL(url);
        this.cache.set(url, texture);
        resolve(texture);
    }
}
```
